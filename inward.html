async function uploadDocument(file, number) { if (!file) return { url: "", name: "" }; const safeName = file.name.replace(/[^a-zA-Z0-9._-]/g, "_"); const path = number + "/" + Date.now() + "_" + safeName; const { error } = await supabaseClient.storage .from("documents") .upload(path, file, { cacheControl: "3600", upsert: true, contentType: file.type || "application/octet-stream" }); if (error) { throw new Error( "Document upload failed: " + error.message ); } const { data } = supabaseClient.storage .from("documents") .getPublicUrl(path); return { url: data.publicUrl, name: file.name }; }
